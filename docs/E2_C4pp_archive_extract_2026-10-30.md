## 5. 复现命令

```bash
unzip -q 交付包_必要性重建_E2_2026-10-11.zip -d e2
cd e2/交付包_必要性重建_E2_2026-10-11/必要性重建/E2_s2c_learnable

import json, glob, statistics as st
rows=[]
for f in sorted(glob.glob('logs/result_seed*.json')):
    d=json.load(open(f,encoding='utf-8')); g=d['arms']['G_RL_GRU']; c=g['C4pp']
    rows.append((d['seed'], c['sparse']['dwf'], c['every']['dwf'],
                 c['every']['dwf']/c['sparse']['dwf'], c['iii_specific'],
                 g['C6']['pass'], g['C1']['pass']))
for r in rows: print('%d  %.4f  %.4f  %.3f  iii=%s C6=%s C1=%s' % r)
r=[x[3] for x in rows]
print('mean %.4f sd %.4f se %.4f' % (st.mean(r), st.stdev(r), st.stdev(r)/len(r)**.5))
PY
```
